#   gh api repos/Dao-AILab/causal-conv1d/releases/tags/v1.6.1.post4 --jq '.assets[].name'
#

#   flash-linear-attention 0.5.0 → fla-core 0.5.0 → torch>=2.7 → cu130 wheels →
